[CCRM-151] Add support for the
OVERLAY linker script command
 
  支持链接脚本语法关键字OVERLAY 。 OVERLAY 主要用于 flash 充足但 ram 不足的场景中，通过
OVERLAY 可以实现将不同的函数或者变量放到同一个 ram 内执行 ( 主要是函数 ) 。因此在每次执行函数或
者需要变量值时，需要将相关数据复制到对应的 ram 区域。因此需要知道函数或者变量的 VMA 和 LMA 。
llvm 中不会自动生成相关符号，关于地址需要用户自行设置。具体测试如下所示：

/**
 * @brief TCARM - 158 被测表达式： ((((A & B) ^ 1) & C) & A)
 */
--- PAGE 6 ---
static inline uint32_t tcarm158_expr(uint32_t A, uint32_t B, uint32_t C)
{
    return ((((A & B) ^ 1) & C) & A);
}
/**
 * @brief TCARM - 158 回归校验
 * @param[in]  input_A    A 输入数组
 * @param[in]  input_B    B 输入数组
 * @param[in]  input_C    C 输入数组
 * @param[in]  expect_out 每组对应的预期输出数组
 * @param[in]  count      测试用例数量
 * @return 失败用例个数； 0= 全部通过； >0= 存在不匹配（旧 bug 编译器会返回非 0 ）
 *
 * 编译要求： -O2 / -O3 ， -O0 不会复现 bug
 */
int tcarm158_verify(const uint32_t *input_A,
                    const uint32_t *input_B,
                    const uint32_t *input_C,
                    const uint32_t *expect_out,
                    unsigned int count)
{
    unsigned int fail_cnt = 0U;
    for (unsigned int i = 0U; i < count; i++)
    {
        uint32_t A = input_A[i];
        uint32_t B = input_B[i];
        uint32_t C = input_C[i];
        uint32_t expect = expect_out[i];
        /* 被测：编译器生成代码，旧版本会出错  */
        uint32_t actual = tcarm158_expr(A, B, C);
        if (actual != expect)
        {
            fail_cnt++;
        }
    }
    return (int)fail_cnt;
}
